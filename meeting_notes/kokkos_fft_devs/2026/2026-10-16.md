## KokkosFFT batched capability https://github.com/kokkos/kokkos-fft/pull/541

### API

```C++
// Serial
KokkosFFT::Batched::Plan plan(exec_space, x, x_hat, Axis0{});

Kokkos::parallel_for(
    Kokkos::RangePolicy<execution_space, Kokkos::IndexType<std::size_t>>{
        exec_space, 0, nbatch},
    KOKKOS_LAMBDA(const std::size_t i) {
      auto sub_x     = Kokkos::subview(x, Kokkos::ALL, i);
      auto sub_x_hat = Kokkos::subview(x_hat, Kokkos::ALL, i);
      // x -> x_hat (x is overwritten)
      KokkosFFT::Batched::execute(plan, sub_x, sub_x_hat, KokkosFFT::Direction::forward);
      // x_hat -> x (x_hat is overwritten)
      KokkosFFT::Batched::execute(plan, sub_x_hat, sub_x, KokkosFFT::Direction::backward);
    });

// Team
KokkosFFT::Batched::Plan plan(team_policy, x, x_hat, Axis0{});
team_policy.set_scratch_size(level, Kokkos::PerTeam(plan.get_scratch_size(level)));

Kokkos::parallel_for(
    "batched_fft_1d", team_policy,
    KOKKOS_LAMBDA(const member_type& team_member) {
      const int i    = team_member.league_rank();
      auto sub_x     = Kokkos::subview(x, Kokkos::ALL, i);
      auto sub_x_hat = Kokkos::subview(x_hat, Kokkos::ALL, i);
      KokkosFFT::Batched::execute(team_member, plan, sub_x, sub_x_hat,
                       KokkosFFT::Direction::forward);
    });
```
